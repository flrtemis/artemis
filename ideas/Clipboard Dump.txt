    let agents = [
      {
        id: "sub-agent-1",
        model: "gpt-neo-1.3B",
        system_prompt: "You specialize in data gathering.",
        weight: 1.0,
        task: "Collect raw information."
      }
    ];

    function renderAgents() {
      const container = document.getElementById("agents");
      container.innerHTML = "";
      agents.forEach((agent, index) => {
        container.innerHTML += `
          <div class="agent">
            <h3>Agent ${index + 1}</h3>
            <label>Model:<br><input type="text" value="${agent.model}" onchange="updateAgent(${index}, 'model', this.value)"></label><br><br>
            <label>System Prompt:<br><textarea onchange="updateAgent(${index}, 'system_prompt', this.value)">${agent.system_prompt}</textarea></label><br>
            <label>Weight:<br><input type="number" step="0.1" value="${agent.weight}" onchange="updateAgent(${index}, 'weight', this.value)"></label><br><br>
            <label>Task:<br><input type="text" value="${agent.task}" onchange="updateAgent(${index}, 'task', this.value)"></label><br>
            <button class="delete" onclick="removeAgent(${index})">🗑 Delete Agent</button>
          </div>
        `;
      });
    }

    function updateAgent(index, key, value) {
      agents[index][key] = value;
    }

    function addAgent() {
      agents.push({
        id: "sub-agent-" + (agents.length + 1),
        model: "gpt-neo-1.3B",
        system_prompt: "New sub-agent prompt",
        weight: 1.0,
        task: "New task"
      });
      renderAgents();
    }

    function removeAgent(index) {
      agents.splice(index, 1);
      renderAgents();
    }

    function exportJSON() {
      const json = JSON.stringify({ main_model: "gpt-4", objective: "Complete delegated task", agents: agents }, null, 2);
      document.getElementById("output").textContent = json;
    }

    renderAgents();
  </script>

import re
import requests

def get_microsoft_store_download_url(share_url: str) -> str:
    """
    Given a Microsoft Store share URL (e.g. https://aka.ms/store/apps/9WZDNCRFJ2XR),
    return the direct download URL (MSIX/MSI) for the *Windows* platform.

    Raises:
        ValueError: If the URL can’t be parsed or if the API response is invalid.
    """
    # 1. Pull out the 16‑character app identifier.
    match = re.search(r"store/apps/([a-zA-Z0-9]{16})", share_url)
    if not match:
        raise ValueError(f"Could not find a valid app ID in {share_url!r}")

    app_id = match.group(1)

    # 2. Prepare a session (re‑uses TCP connections & handles cookies).
    session = requests.Session()
    session.headers.update(
        {
            "User-Agent": "Mozilla/5.0",
            "Accept": "application/json",
        }
    )

    # 3. Call the Store content API.
    api_url = f"https://storecontent.microsoft.com/api/content/{app_id}"
    params = {"locale": "en-US", "platform": "Windows"}
    response = session.get(api_url, params=params)

    # 4. Basic HTTP error handling.
    if response.status_code != 200:
        raise ValueError(
            f"Store API returned status {response.status_code} for {app_id}"
        )

    # 5. Parse JSON safely.
    try:
        data = response.json()
    except ValueError as exc:
        # Not JSON – maybe a CAPTCHA page or HTML error.
        raise ValueError(
            f"Failed to parse JSON from the Store API for {app_id}"
        ) from exc

    # 6. Extract the download URL if present.
    download_url = data.get("DownloadUrl")
    if not download_url:
        raise ValueError(
            f"The store API response for {app_id} did not contain a DownloadUrl"
        )

    return download_url

import ollama
import sys
import os

def read_file(file_path):
    """Read the content of a file"""
    try:
        with open(file_path, 'r', encoding='utf-8') as file:
            content = file.read()
        return content
    except FileNotFoundError:
        print(f"Error: File '{file_path}' not found.")
        sys.exit(1)
    except Exception as e:
        print(f"Error reading file: {e}")
        sys.exit(1)

def call_ollama_api(file_content, file_path, model="qwen3-coder:30b"):
    """Call Ollama API with the file content and a specific instruction for code analysis."""
    
    # This system prompt is crucial. It tells the specialized coding model what its task is.
    system_prompt = (
        "You are an expert Python programmer. The user will provide the complete contents of a Python file. "
        "Your task is to analyze the code, explain its purpose, identify any potential issues (bugs, inefficiencies, bad practices), "
        "and suggest improvements. If the code is incomplete, suggest how to complete it. "
        "Focus on being helpful and actionable. Format your response clearly."
    )
    
    # Construct the full prompt for the model, providing crucial context.
    full_prompt = f"<|im_start|>system\n{system_prompt}<|im_end|>\n"
    full_prompt += f"<|im_start|>user\nAnalyze the following Python file: '{file_path}'\n\n```python\n{file_content}\n```<|im_end|>\n"
    full_prompt += "<|im_start|>assistant\n"

    try:
        # Use the chat API instead of generate for structured conversation
        response = ollama.chat(
            model=model,
            messages=[
                {'role': 'system', 'content': system_prompt},
                {'role': 'user', 'content': f"Analyze this Python file: {file_path}\n\n{file_content}"}
            ],
            stream=False,
            options={
                'temperature': 0.1, # Lower temperature for more deterministic, code-focused output
                'num_ctx': 8192     # You can try increasing the context window
            }
        )
        return response['message']['content']
    except Exception as e:
        print(f"Error calling Ollama API: {e}")
        sys.exit(1)

def main():
    if len(sys.argv) != 2:
        print("Usage: python analyze_code.py <path_to_python_file.py>")
        sys.exit(1)

    file_path = sys.argv[1]

    if not os.path.exists(file_path):
        print(f"Error: File '{file_path}' does not exist.")
        sys.exit(1)
        
    if not file_path.endswith('.py'):
        print("Warning: This tool is designed for Python files (.py).")

    print(f"[*] Reading file: {os.path.abspath(file_path)}")
    content = read_file(file_path)
    print(f"[*] File read successfully. Sending to Qwen3-Coder for analysis...")

    response = call_ollama_api(content, file_path)

    print("\n" + "="*70)
    print(f"ANALYSIS FOR: {file_path}")
    print("="*70)
    print(response)
    print("="*70)

if __name__ == "__main__":
    main()

import sqlite3
import subprocess
import json
import numpy as np
import os
from PyPDF2 import PdfReader

DB = "docs.db"

# --- DB setup ---
conn = sqlite3.connect(DB)
cur = conn.cursor()
cur.execute("""
CREATE TABLE IF NOT EXISTS documents (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  content TEXT,
  embedding BLOB
)
""")
conn.commit()

# --- Embedding ---
def get_embedding(text: str):
    result = subprocess.run(
        ["ollama", "embed", "snowflake-arctic-embed:22m", "-p", text],
        capture_output=True, text=True
    )
    data = json.loads(result.stdout)
    return np.array(data["embedding"], dtype=np.float32)

# --- Add doc ---
def add_document(content: str):
    emb = get_embedding(content)
    cur.execute("INSERT INTO documents (content, embedding) VALUES (?, ?)", 
                (content, emb.tobytes()))
    conn.commit()

# --- Load text file ---
def load_txt(path: str):
    with open(path, "r", encoding="utf-8") as f:
        return f.read()

# --- Load pdf ---
def load_pdf(path: str):
    reader = PdfReader(path)
    text = ""
    for page in reader.pages:
        text += page.extract_text() + "\n"
    return text

# --- Ingest file ---
def add_file(path: str):
    ext = os.path.splitext(path)[1].lower()
    if ext == ".txt":
        text = load_txt(path)
    elif ext == ".pdf":
        text = load_pdf(path)
    else:
        print("Unsupported file type:", ext)
        return

    # simple chunking
    chunks = [text[i:i+500] for i in range(0, len(text), 500)]
    for chunk in chunks:
        if chunk.strip():
            add_document(chunk)
    print(f"File {path} ingested with {len(chunks)} chunks.")

# --- Search ---
def search(query: str, top_k=3):
    q_emb = get_embedding(query)
    cur.execute("SELECT id, content, embedding FROM documents")
    rows = cur.fetchall()

    results = []
    for rid, content, emb_blob in rows:
        emb = np.frombuffer(emb_blob, dtype=np.float32)
        sim = np.dot(q_emb, emb) / (np.linalg.norm(q_emb) * np.linalg.norm(emb))
        results.append((sim, content))

    results.sort(reverse=True, key=lambda x: x[0])
    return results[:top_k]

# --- Generate with context ---
def ask_with_context(query: str, top_k=3):
    matches = search(query, top_k)
    context = "\n".join([c for _, c in matches])
    prompt = f"Use the following context to answer:\n{context}\n\nQuestion: {query}"

    result = subprocess.run(
        ["ollama", "run", "llama3:latest"],
        input=prompt,
        text=True,
        capture_output=True
    )
    return result.stdout.strip()

# --- REPL ---
if __name__ == "__main__":
    print("Interactive RAG REPL. Commands:")
    print("  add <text>")
    print("  ask <question>")
    print("  file <path-to-.txt-or-.pdf>")
    print("  quit")

    while True:
        cmd = input(">> ").strip()
        if cmd.lower() in ["quit", "exit"]:
            break
        elif cmd.startswith("add "):
            content = cmd[4:]
            add_document(content)
            print("Document added.")
        elif cmd.startswith("ask "):
            q = cmd[4:]
            answer = ask_with_context(q)
            print("Answer:", answer)
        elif cmd.startswith("file "):
            path = cmd[5:].strip()
            if os.path.exists(path):
                add_file(path)
            else:
                print("File not found:", path)
        else:
            print("Unknown command. Use 'add', 'ask', or 'file'.")

def read_file(file_path):
    """Read the content of a file"""
    try:
        with open(file_path, 'r', encoding='utf-8') as file:
            content = file.read()
        return content
    except FileNotFoundError:
        print(f"Error: File '{file_path}' not found.")
        sys.exit(1)
    except Exception as e:
        print(f"Error reading file: {e}")
        sys.exit(1)

def call_ollama_api(file_content, file_path, model="qwen3-coder:30b"):
    """Call Ollama API with the file content and a specific instruction for code analysis."""
    
    # This system prompt is crucial. It tells the specialized coding model what its task is.
    system_prompt = (
        "You are an expert Python programmer. The user will provide the complete contents of a Python file. "
        "Your task is to analyze the code, explain its purpose, identify any potential issues (bugs, inefficiencies, bad practices), "
        "and suggest improvements. If the code is incomplete, suggest how to complete it. "
        "Focus on being helpful and actionable. Format your response clearly."
    )
    
    # Construct the full prompt for the model, providing crucial context.
    full_prompt = f"<|im_start|>system\n{system_prompt}<|im_end|>\n"
    full_prompt += f"<|im_start|>user\nAnalyze the following Python file: '{file_path}'\n\n```python\n{file_content}\n```<|im_end|>\n"
    full_prompt += "<|im_start|>assistant\n"

    try:
        # Use the chat API instead of generate for structured conversation
        response = ollama.chat(
            model=model,
            messages=[
                {'role': 'system', 'content': system_prompt},
                {'role': 'user', 'content': f"Analyze this Python file: {file_path}\n\n{file_content}"}
            ],
            stream=False,
            options={
                'temperature': 0.1, # Lower temperature for more deterministic, code-focused output
                'num_ctx': 8192     # You can try increasing the context window
            }
        )
        return response['message']['content']
    except Exception as e:
        print(f"Error calling Ollama API: {e}")
        sys.exit(1)

def main():
    if len(sys.argv) != 2:
        print("Usage: python analyze_code.py <path_to_python_file.py>")
        sys.exit(1)

    file_path = sys.argv[1]

    if not os.path.exists(file_path):
        print(f"Error: File '{file_path}' does not exist.")
        sys.exit(1)
        
    if not file_path.endswith('.py'):
        print("Warning: This tool is designed for Python files (.py).")

    print(f"[*] Reading file: {os.path.abspath(file_path)}")
    content = read_file(file_path)
    print(f"[*] File read successfully. Sending to Qwen3-Coder for analysis...")

    response = call_ollama_api(content, file_path)

    print("\n" + "="*70)
    print(f"ANALYSIS FOR: {file_path}")
    print("="*70)
    print(response)
    print("="*70)

if __name__ == "__main__":
    main()

import ollama
import sys
import os

def read_file(file_path):
    """Read the content of a file"""
    try:
        with open(file_path, 'r', encoding='utf-8') as file:

    except FileNotFoundError:
        print(f"Error: File '{file_path}' not found.")
        sys.exit(1)
    except Exception as e:

        sys.exit(1)

def call_ollama_api(content, model="llama3", host="http://localhost:11434"):
    """Call Ollama API with the file content"""
    try:
        response = ollama.generate(
            model=model,
            prompt=content,
            stream=False
        )
        return response['response']

        print(f"Error calling Ollama API: {e}")

```python
import ollama
import sys
import os

def read_file(file_path):
    """Read the content of a file"""
    try:
        with open(file_path, 'r', encoding='utf-8') as file:
            content = file.read()
        return content
    except FileNotFoundError:
        print(f"Error: File '{file_path}' not found.")
        sys.exit(1)
    except Exception as e:
        print(f"Error reading file: {e}")
        sys.exit(1)

def call_ollama_api(content, model="llama3", host="http://localhost:11434", options=None):
    """Call Ollama API with the file content"""
    try:
        response = ollama.generate(
            model=model,
            prompt=content,
            stream=False,
            options=options or {}
        )
        return response['response']
    except Exception as e:
        print(f"Error calling Ollama API: {e}")
        sys.exit(1)

def main():
    # Default parameters
    model = "llama3"
    host = "http://localhost:11434"
    options = None

    # Check if file path is provided as command line argument
    if len(sys.argv) < 2:
        print("Usage: python script.py <file_path> [model_name]")
        sys.exit(1)

    file_path = sys.argv[1]

    # Validate file exists
    if not os.path.exists(file_path):
        print(f"Error: File '{file_path}' does not exist.")
        sys.exit(1)

    # Optional model specification
    if len(sys.argv) > 2:
        model = sys.argv[2]

    # Read file content
    print(f"Reading file: {file_path}")
    content = read_file(file_path)

    # Call Ollama API
    print(f"Sending content to Ollama API (model: {model})...")
    response = call_ollama_api(content, model=model, host=host, options=options)

    # Display results
    print("\n" + "="*50)
    print("Ollama API Response:")
    print("="*50)
    print(response)
    print("="*50)

if __name__ == "__main__":
    main()
```

- Reads any text file and sends its content to Ollama

- Supports custom model selection
- Provides clear output formatting
- Works with the default Ollama installation on localhost
